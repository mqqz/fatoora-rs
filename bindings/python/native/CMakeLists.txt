cmake_minimum_required(VERSION 3.19)
project(fatoora_python LANGUAGES CXX)
find_package(Python 3.9 COMPONENTS Interpreter Development.Module REQUIRED)
execute_process(COMMAND "${Python_EXECUTABLE}" -m nanobind --cmake_dir
    OUTPUT_VARIABLE nanobind_DIR OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
find_package(nanobind 2.12.0 EXACT CONFIG REQUIRED)

file(GLOB_RECURSE binding_sources CONFIGURE_DEPENDS generated/*.cpp)
nanobind_add_module(_native ${binding_sources})
target_compile_features(_native PRIVATE cxx_std_20)
target_include_directories(_native PRIVATE generated/include)
# Diplomat's deleter shim includes Nanobind internals, which depend on robin_map.
target_include_directories(_native PRIVATE "${NB_DIR}/ext/robin_map/include")
add_library(fatoora_ffi SHARED IMPORTED)
set_target_properties(fatoora_ffi PROPERTIES IMPORTED_LOCATION "${FATOORA_LIBRARY}")
if(UNIX AND NOT APPLE)
    # Rust cdylibs have no SONAME by default. Avoid an absolute DT_NEEDED path.
    set_target_properties(fatoora_ffi PROPERTIES IMPORTED_NO_SONAME TRUE)
endif()
if(WIN32)
    set_target_properties(fatoora_ffi PROPERTIES IMPORTED_IMPLIB "${FATOORA_IMPLIB}")
endif()
target_link_libraries(_native PRIVATE fatoora_ffi)
if(APPLE)
    set_target_properties(_native PROPERTIES INSTALL_RPATH "@loader_path")
elseif(UNIX)
    set_target_properties(_native PROPERTIES INSTALL_RPATH "$ORIGIN")
endif()
install(TARGETS _native LIBRARY DESTINATION fatoora RUNTIME DESTINATION fatoora)
